\section{Optimization and Lagrange Multipliers}
\subsection{Extrema of Multivariable Functions}

\begin{definition}{Local Maximum and Minimum}{local_max_min}
For a function $f(x, y)$ and a point $(a, b)$ in its domain, we say that $(a, b)$ is a 
\begin{itemize}
    \item \textbf{local maximum} if $f(a, b) \geq f(x, y),\ \forall (x, y)$ in a neighborhood of $(a, b)$.
    \item \textbf{local minimum} if $f(a, b) \leq f(x, y),\ \forall (x, y)$ in a neighborhood of $(a, b)$.
\end{itemize}
\end{definition}

\begin{definition}{Critical Point}{critical_point}
We call a point $P(a, b)$ a \textbf{critical point} of $f$ if
\[
\nabla f(P) = \vv{0}
\]
\end{definition}

\begin{theorem}{Critical Points and Local Extrema}{critical_points_local_extrema}
If $P(a, b)$ is a local extremum of $f$, then $P$ is a critical point of $f$.
\end{theorem}

For single variable functions, we can use the second derivative test to determine the nature of a point on the function.
For $f(x)$, if $f'(a) = 0$, then:
\begin{itemize}
    \item If $f''(a) > 0$, then $f$ has a local minimum at $x = a$.
    \item If $f''(a) < 0$, then $f$ has a local maximum at $x = a$.
    \item If $f''(a) = 0$, the test is inconclusive.
\end{itemize}

For multivariable functions, we can use its quadratic approximation to determine the nature of a critical point.

\begin{problem}{Determining the Critical Points of a Function}{determine_critical_points}
Find critical points of the function $f(x, y) = x^4 + y^4 + 4xy$. Which are local maxima, local minima, and saddle points?
\end{problem}

\begin{sol}
First, we find the critical points by setting the gradient of $f$ to zero. The gradient is given by:
\[
\nabla f(x, y) = \ang{\pdv{f}{x}, \pdv{f}{y}} = (4x^3 + 4y, 4y^3 + 4x)
\]
Setting each component to zero gives us the system of equations:
\begin{align*}
4x^3 + 4y &= 0 \\
4y^3 + 4x &= 0
\end{align*}

This implies $y = -x^3$. Substituting $y = -x^3$ into the second equation gives:
\[
4(-x^3)^3 + 4x = 0 \implies -4x^9 + 4x = 0 \implies 4x(-x^8 + 1) = 0
\]
This gives us $x = 0$ or $x^8 = 1$. The equation $x^8 = 1$ has solutions $x = \pm 1$. So the critical points are $(0, 0)$, $(1, -1)$, and $(-1, 1)$.

To determine the nature of these critical points, we write $f(x, y)$ in the quadratic approximation form around each critical point.
\begin{align*}
f(x, y) &\approx a + bx + cy + Ax^2 + Bxy + Cy^2 \\
\text{If } \nabla f(0, 0) &= 0, \text{ then } b = c = 0, a = f(0, 0) \\
\implies f(x, y) &\approx f(0, 0) + Ax^2 + Bxy + Cy^2
\end{align*}
The second derivative is then given by
\begin{align*}
\pdv{f}{x}{x} &= 2A \\
\pdv{f}{y}{y} &= 2C \\
\pdv{f}{x}{y} &= B
\end{align*}
\end{sol}

Fact: If $(a, b)$ is a critical point of $f$, then the quadratic approximation is given by
\[
Q(x, y) = \frac{1}{2} f_{xx}(a, b)(x - a)^2 + f_{xy}(a, b)(x - a)(y - b) + \frac{1}{2} f_{yy}(a, b)(y - b)^2
\]
There are three possible nondegenerate behaviors for a quadratic function near a critical point:

For a general quadratic function
\[
Q(x,y)=Ax^2 + Bxy + Cy^2,
\]
its behavior is determined by $4AC - B^2$. For the quadratic approximation of $f$ at a critical point $(a,b)$,
\[
A=\frac12 f_{xx}(a, b),\qquad
B=f_{xy}(a, b),\qquad
C=\frac12 f_{yy}(a, b).
\]
Hence,
\[
4AC-B^2=f_{xx}(a, b)f_{yy}(a, b)-f_{xy}(a, b)^2.
\]

\begin{definition}{Second Derivative Test}{second_derivative_test}
Let $(a,b)$ be a critical point of $f$, and define
\[
D=f_{xx}(a, b)f_{yy}(a, b)-f_{xy}(a, b)^2.
\]
Then:
\begin{itemize}
    \item If $D > 0$ and $f_{xx}(a, b)>0$, then $(a, b)$ is a \textbf{local minimum}.
    \item If $D > 0$ and $f_{xx}(a, b)<0$, then $(a, b)$ is a \textbf{local maximum}.
    \item If $D < 0$, then $(a, b)$ is a \textbf{saddle point}.
    \item If $D = 0$, the test is \textbf{inconclusive}.
\end{itemize}
\end{definition}

\begin{example}[Usage of the Discriminant]
We have $f(x, y) = x^2 + y^2 - 3xy$. 

From calculation, we know that $\nabla f = \ang{2x - 3y, 2y - 3x}$.
Setting this equal to zero gives us the system of equations:
\[
2x - 3y = 0 \quad \text{and} \quad 2y - 3x = 0.
\]
Solving this system, we find that the only critical point is $(0, 0)$.

The discriminant is given by $D = (2)(2) - (-3)^2 = 4 - 9 = -5 < 0$.

We now know that $(0, 0)$ is a saddle.

\end{example}

\begin{definition}{Global Extrema}{global_extrema}
Given a function $f(x, y)$ defined on some domain $\mathcal{D}$, we say that $(a, b) \in \mathcal{D}$ is a
\begin{itemize}
    \item \textbf{global maximum} if $f(a, b) \geq f(x, y),\ \forall (x, y) \in \mathcal{D}$.
    \item \textbf{global minimum} if $f(a, b) \leq f(x, y),\ \forall (x, y) \in \mathcal{D}$.
\end{itemize}
\end{definition}

\subsection{Domains and Boundaries}


\begin{definition}{Interior and Boundary Points}{interior_boundary_points}
Let $\mathcal{D} \subseteq \mathbb{R}^2$ be a domain.
\begin{itemize}
    \item A point $(a, b) \in \mathcal{D}$ is an \textbf{interior point} if is has some disk around it in $\mathcal{D}$.
    \item A point $(a, b) \in \mathcal{D}$ is a \textbf{boundary point} such that every disk around it intersects $\mathcal{D}$ and $\mathbb{R}^2 \setminus \mathcal{D}$.
\end{itemize}
* $\mathbb{R}^2 \setminus D$ means the set of points in $\mathbb{R}^2$ that are not in $D$.
\end{definition}

\begin{definition}{Closed and Bounded Sets}{closed_bounded_sets}
A set $\mathcal{D} \subseteq \mathbb{R}^2$ is \textbf{closed} if it contains all of its boundary points,
and \textbf{bounded} if it can be contained in a disk of finite radius.
\end{definition}

\begin{theorem}{Extreme Value Theorem}{extreme_value_theorem}
Let $f$ be a continuous function on a closed and bounded set $\mathcal{D} \subseteq \mathbb{R}^2$.

Then $f$ attains both a global maximum and a global minimum on $\mathcal{D}$.
\end{theorem}

\begin{example}[Finding Global Extrema]
Given the function $f(x, y) = x^2 + y^2 + xy$, and the domain $D = \{(x, y) : x^2 + y^2 \leq 4\}$.

We set the gradient of $f$ to zero to find critical points:
\[
\nabla f = \ang{2x + y, 2y + x} = \ang{0, 0}.
\]
This gives us the system of equations:
\[
2x + y = 0 \quad \text{and} \quad 2y + x = 0.
\]
Solving this system, we find that the only critical point is $(0, 0)$.

Since $\mathcal{D}$ is closed and bounded, and $f$ is continuous on $\mathcal{D}$, the Extreme Value Theorem guarantees that $f$ attains both a global maximum and a global minimum on $\mathcal{D}$.
Therefore, we need to check the boundary of $\mathcal{D}$ to find the global extrema.

The boundary of $\mathcal{D}$ is the circle defined by $x^2 + y^2 = 4$. We can parameterize this boundary using polar coordinates:
\[
x = 2\cos(t), \quad y = 2\sin(t), \quad 0 \leq t < 2\pi.
\]
Hence,
\begin{align*}
f(2\cos(t), 2\sin(t)) &= (2\cos(t))^2 + (2\sin(t))^2 + (2\cos(t))(2\sin(t)) \\
&= 4\cos^2(t) + 4\sin^2(t) + 4\cos(t)\sin(t) \\
&= 4 + 4\cos(t)\sin(t) \\
&= 4 + 2\sin(2t).
\end{align*}
Therefore, the maximum value of $f$ on the boundary is $4 + 2 = 6$ at $x = 2, y = 0$, and the minimum value is $4 - 2 = 2$ at $x = -2, y = 0$.
\end{example}

\begin{theorem}{Lagrange Multipliers}{lagrange_multipliers}
Let $f(x, y)$ be a function and $g(x, y) = c$ be a constraint curve.
If $g(x, y) = c$ is tangent to a level curve of $f$, then
\[
    \nabla f \parallel \nabla g \implies \nabla f = \lambda \nabla g
\] 

Therefore, to find the extrema of $f$ under the constraint $g(x, y) = c$, we can solve the system of equations:
\[
    \nabla f = \lambda \nabla g, \quad g(x, y) = c
\]
for $x$, $y$, and $\lambda$, assuming that $\nabla g \neq 0$ at the points of interest.
\end{theorem}

\begin{example}[Using Lagrange Multipliers]
We want to find the extrema of $f(x, y) = x + 2y$ subject to the constraint $3x^2 + 2y^2 = 7$.

First, we compute the gradients:
\[
\nabla f = \ang{1, 2}, \quad \nabla g = \ang{6x, 4y}.
\]
Setting $\nabla f = \lambda \nabla g$ gives us the system of equations:
\[
    \left\{
    \begin{aligned}
        &1 = \lambda(6x)\\
        &2 = \lambda(4y)\\
        &3x^2 + 2y^2 = 7
    \end{aligned}
    \right.
\]

Solving the first two equations for $\lambda$, we get:
\[
    \left\{
    \begin{aligned}
        \lambda &= \dfrac{1}{6x} \\
        \lambda &= \dfrac{1}{2y}
    \end{aligned}
    \right.
\]
Setting these equal gives us:
\[
    \dfrac{1}{6x} = \dfrac{1}{2y} \implies 2y = 6x \implies y = 3x.
\]

Substituting $y = 3x$ into the constraint equation gives:
\begin{align*}
    3x^2 + 2(3x)^2 &= 7 \\
    3x^2 + 18x^2 &= 7 \\
    21x^2 &= 7 \\
    x^2 &= \frac{1}{3} \\
    x &= \pm \frac{1}{\sqrt{3}}.
\end{align*}
Thus, the corresponding $y$ values are:
\[
    y = 3\left(\pm \frac{1}{\sqrt{3}}\right) = \pm \sqrt{3}.
\]
Therefore, the points of interest are:
\[\left(\frac{1}{\sqrt{3}}, \sqrt{3}\right) \text{ and } \left(-\frac{1}{\sqrt{3}}, -\sqrt{3}\right).\]
\end{example}

\begin{example}[Another Example of Lagrange Multipliers]
Given a function $Q(K, L) = A K^\alpha L^{1-\alpha}$.
This describes the output when $K$ units of capital and $L$ units of labor are used, where $A > 0$ is a constant and $0 < \alpha < 1$.

Suppose that $A = 1$ and $\alpha = \frac{3}{5}$. Given a budget of $\mathdollar 12$ with the cost of capital being $\mathdollar 3$ per unit and the cost of labor being $\mathdollar 4$ per unit, we want to maximize $Q(K, L)$ subject to the constraint $2K + 3L = 12$.

The process is similar to the previous example. We first rewrite the constraint as $g(K, L) = 2K + 3L - 12 = 0$. Then, we compute the gradients:
\begin{align*}
    \nabla Q &= \ang{\frac{3}{5} K^{-\nicefrac{2}{5}} L^{\nicefrac{2}{5}}, \frac{2}{5} K^{\nicefrac{3}{5}} L^{-\nicefrac{3}{5}}} \\
    \nabla g &= \ang{2, 3}.
\end{align*}
Setting $\nabla Q = \lambda \nabla g$ gives us the system of equations:
\[
    \left\{
    \begin{aligned}
        &\frac{3}{5} K^{-\nicefrac{2}{5}} L^{\nicefrac{2}{5}} = 2\lambda \\
        &\frac{2}{5} K^{\nicefrac{3}{5}} L^{-\nicefrac{3}{5}} = 3\lambda \\
        &2K + 3L = 12
    \end{aligned}
    \right.
\]
Solving the first two equations for $\lambda$, we get:
\[
    \frac{3}{10} K^{-\nicefrac{2}{5}} L^{\nicefrac{2}{5}} = \frac{2}{15} K^{\nicefrac{3}{5}} L^{-\nicefrac{3}{5}} \implies 9L = 4K \implies L = \frac{4}{9}K.
\]
Substituting $L = \frac{4}{9}K$ into the constraint equation gives:
\begin{align*}
    2K + 3\left(\frac{4}{9}K\right) &= 12 \\[0.5em]
    2K + \frac{4}{3}K &= 12 \\[0.5em]
    \frac{10}{3}K &= 12 \\[0.5em]
    K &= \frac{36}{10} = 3.6.
\end{align*}
Then, substituting $K = 3.6$ into $L = \dfrac{4}{9}K$ gives:
\[
    L = \frac{4}{9} \cdot 3.6 = 1.6.
\]
Therefore, the optimal allocation of resources is $K = 3.6$ units of capital and $L = 1.6$ units of labor.
\end{example}
