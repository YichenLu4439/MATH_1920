\section{Double Integrals}
\begin{definition}{Imprecise Definition of Double Integral}{def_double_integral_imprecise}
Let $f(x, y)$ be a function defined on a closed and bounded region $\mathcal{D}$ in the $xy$-plane.
The \textbf{double integral} is defined by
\[
    \dint{f(x, y)}
\]
where $\dd{A}$ is the area element in the $xy$-plane.

This integral can be interpreted as the volume under the surface $z = f(x, y)$ and above the region $\mathcal{D}$ in the $xy$-plane.
\end{definition}

Below are some properties of double integrals that will be useful in evaluating them.
\begin{itemize}
    \item $\displaystyle\dint{f(x, y) + g(x, y)} = \dint{f(x, y)} + \dint{g(x, y)}$ \\[-0.5em]
    \item $\displaystyle\dint{c \cdot f(x, y)} = c \cdot \dint{f(x, y)}$ \\[-0.5em]
    \item If $f(x, y) \leq g(x, y)$ for all $(x, y) \in \mathcal{D}$, then $\displaystyle\dint{f(x, y)} \leq \dint{g(x, y)}$.
\end{itemize}

By this, we could write a precise definition of the double integral as follows:
\begin{definition}{Precise Definition of Double Integral}{def_double_integral_precise}
Let $f(x, y)$ be a function defined on a closed and bounded region $\mathcal{D}$ in the $xy$-plane.

We partition $\mathcal{D}$ into $m$ subintervals in the $x$-direction and $n$ subintervals in the $y$-direction, forming a grid of rectangles.

Let $\Delta x_i$ and $\Delta y_j$ be the widths of the subintervals in the $x$ and $y$ directions, respectively,
and let $P_{ij}$ be a sample point in the rectangle formed by the $i$-th subinterval in the $x$-direction and the $j$-th subinterval in the $y$-direction.

The \textbf{double integral} of $f(x, y)$ over the region $\mathcal{D}$ is defined as the limit of the Riemann sum as the maximum width of the subintervals approaches zero:
\[
    \dint{f(x, y)} = \lim_{\norm{P} \to 0} \sum_{i=1}^m \sum_{j=1}^n f(P_{ij}) \Delta A_{ij}
\]
\end{definition}

Notice that the $\Delta A_{ij}$ here does not always exist. If it does, we call the function $f(x, y)$ \textbf{integrable} over the region $\mathcal{D}$.
Also, if a function is continuous over a closed and bounded region, then it is integrable over that region.

\subsection{Iterated Integrals and Fubini's Theorem}
To evaluate double integrals, we can use Fubini's Theorem, which allows us to compute the double integral as an iterated integral.
\begin{theorem}{Fubini's Theorem}{thm_fubini}
\[
    \dint{f(x, y)} = \int_a^b \int_c^d f(x, y) \, \dd{y} \, \dd{x} = \int_c^d \int_a^b f(x, y) \, \dd{x} \, \dd{y}
\]
\end{theorem}

Using this theorem, we can evaluate a double integral by computing two single-variable integrals in succession.

\begin{example}[Evaluating a Double Integral]

We first approximate $\dint{f(x, y)}$ using a midpoint Riemann sum.
Divide $[0, 4]$ into two equal intervals and $[0, 6]$ into three equal intervals.
This gives six rectangles, each with area $\Delta A = 2 \cdot 2 = 4$.
The sample points are their centers:
\[(1, 1),\ (1, 3),\ (1, 5),\ (3, 1),\ (3, 3),\ (3, 5).\]

Thus the midpoint Riemann sum is
\begin{align*}
    S &= \sum_{i=1}^{6} f(x_i, y_i) \Delta A \\
      &= 4\bigl(0 - 2 - 4 + 8 + 6 + 4\bigr) = 48.
\end{align*}
The negative terms represent contributions below the $xy$-plane.

By Fubini's theorem, the exact value is
\begin{align*}
    \dint{f(x, y)}
    &= \int_0^4 \int_0^6 (x^2 - y) \, \dd{y} \, \dd{x} \\
    &= \int_0^4 \left[ x^2y - \frac{y^2}{2} \right]_0^6 \, \dd{x} \\
    &= \int_0^4 (6x^2 - 18) \, \dd{x} \\
    &= \left[ 2x^3 - 18x \right]_0^4 = \boxed{56}.
\end{align*}
\end{example}

For vertical slices, fix $x$ and let $y$ run from the lower boundary $y = x^2$ to the upper boundary $y = 1$.
The slices cover the region as $x$ runs from $-1$ to $1$, so
\[\mathcal{D} = \{(x, y) \mid -1 \leq x \leq 1,\ x^2 \leq y \leq 1\}.\]
Hence
\[\dint{f(x, y)} = \int_{-1}^{1} \int_{x^2}^{1} f(x, y) \, \dd{y} \, \dd{x}.\]

For horizontal slices, fix $y$ and solve $x^2 \leq y$ to obtain $-\sqrt{y} \leq x \leq \sqrt{y}$.
The slices cover the region as $y$ runs from $0$ to $1$, so
\[\mathcal{D} = \{(x, y) \mid 0 \leq y \leq 1,\ -\sqrt{y} \leq x \leq \sqrt{y}\}.\]
Thus the other order of integration is
\[\dint{f(x, y)} = \int_0^1 \int_{-\sqrt{y}}^{\sqrt{y}} f(x, y) \, \dd{x} \, \dd{y}.\]
\end{example}

\begin{definition}{Simple Region}{def_simple_region}
A region $\mathcal{D}$ is
\begin{itemize}
    \item \textbf{Vertically simple} if
    \[\mathcal{D} = \{(x, y) \mid a \leq x \leq b,\ g_1(x) \leq y \leq g_2(x)\},\]
    where $g_1$ and $g_2$ are continuous on $[a, b]$ and $g_1(x) \leq g_2(x)$.
    Each vertical line intersects $\mathcal{D}$ in a single line segment, a point, or not at all.

    \item \textbf{Horizontally simple} if
    \[\mathcal{D} = \{(x, y) \mid c \leq y \leq d,\ h_1(y) \leq x \leq h_2(y)\},\]
    where $h_1$ and $h_2$ are continuous on $[c, d]$ and $h_1(y) \leq h_2(y)$.
    Each horizontal line intersects $\mathcal{D}$ in a single line segment, a point, or not at all.
\end{itemize}
\end{definition}

If $f$ is continuous on the corresponding simple region, then
\begin{align*}
    \text{Vertically simple:}\quad
    \dint{f(x, y)} &= \int_a^b \int_{g_1(x)}^{g_2(x)} f(x, y) \, \dd{y} \, \dd{x}, \\
    \text{Horizontally simple:}\quad
    \dint{f(x, y)} &= \int_c^d \int_{h_1(y)}^{h_2(y)} f(x, y) \, \dd{x} \, \dd{y}.
\end{align*}
